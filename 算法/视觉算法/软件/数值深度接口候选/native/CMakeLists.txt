cmake_minimum_required(VERSION 3.16)
project(agtc_depth_native LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(agtc_depth STATIC depth_encode.cpp depth_filter.cpp depth_store.cpp depth_producer.cpp depth_request.cpp depth_http.cpp)
target_compile_features(agtc_depth PUBLIC cxx_std_17)
target_link_libraries(agtc_depth PUBLIC Threads::Threads)
target_include_directories(agtc_depth PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
add_executable(test_depth_store test_depth_store.cpp)
target_link_libraries(test_depth_store PRIVATE agtc_depth)
add_test(NAME depth_store COMMAND test_depth_store)
add_executable(test_depth_pipeline test_depth_pipeline.cpp)
target_link_libraries(test_depth_pipeline PRIVATE agtc_depth)
add_test(NAME depth_pipeline COMMAND test_depth_pipeline)
add_executable(test_depth_http test_depth_http.cpp)
target_link_libraries(test_depth_http PRIVATE agtc_depth)
add_test(NAME depth_http COMMAND test_depth_http)
add_executable(test_http_fixture test_http_fixture.cpp)
target_link_libraries(test_http_fixture PRIVATE agtc_depth)
if(WIN32)
    target_link_libraries(test_http_fixture PRIVATE ws2_32)
endif()

option(AGTC_WITH_OPENCV "Build the native stereo candidate (OpenCV development files required)" OFF)
option(AGTC_WITH_GSTREAMER "Build the GStreamer capture adapter" OFF)
option(AGTC_WITH_HTTP_SERVER "Build the loopback HTTP endpoint" OFF)
option(AGTC_WITH_PREVIEW "Build the fixed-scale JPEG preview" OFF)
set(AGTC_GST_ROOT "" CACHE PATH "Optional GStreamer development prefix")
set(AGTC_JPEG_ROOT "" CACHE PATH "Optional libjpeg-turbo development prefix")
set(AGTC_HTTPLIB_INCLUDE_DIR "" CACHE PATH "Directory containing the pinned cpp-httplib header")
if(AGTC_WITH_GSTREAMER AND NOT AGTC_WITH_OPENCV)
    message(FATAL_ERROR "AGTC_WITH_GSTREAMER requires AGTC_WITH_OPENCV")
endif()
if(AGTC_WITH_PREVIEW AND (NOT AGTC_WITH_OPENCV OR NOT AGTC_WITH_HTTP_SERVER))
    message(FATAL_ERROR "AGTC_WITH_PREVIEW requires AGTC_WITH_OPENCV and AGTC_WITH_HTTP_SERVER")
endif()
if(AGTC_WITH_OPENCV)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc)
    add_library(agtc_stereo STATIC stereo_sgbm.cpp stereo_publisher.cpp stereo_pair_queue.cpp)
    target_compile_features(agtc_stereo PUBLIC cxx_std_17)
    target_include_directories(agtc_stereo PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(agtc_stereo PUBLIC agtc_depth opencv_core opencv_imgproc)
    if(OpenCV_VERSION_MAJOR GREATER_EQUAL 5)
        target_link_libraries(agtc_stereo PUBLIC opencv_stereo opencv_geometry)
    else()
        target_link_libraries(agtc_stereo PUBLIC opencv_calib3d)
    endif()
    add_executable(stereo_replay_fixture stereo_replay_fixture.cpp)
    target_link_libraries(stereo_replay_fixture PRIVATE agtc_stereo)
    add_executable(test_stereo_pair_queue test_stereo_pair_queue.cpp)
    target_link_libraries(test_stereo_pair_queue PRIVATE agtc_stereo)
    add_test(NAME stereo_pair_queue COMMAND test_stereo_pair_queue)
    if(AGTC_WITH_GSTREAMER)
        add_library(agtc_gst STATIC gst_stereo_capture.cpp vision_runtime.cpp)
        target_compile_features(agtc_gst PUBLIC cxx_std_17)
        target_link_libraries(agtc_gst PUBLIC agtc_stereo)
        if(AGTC_GST_ROOT)
            find_path(AGTC_GST_INCLUDE gst/gst.h PATHS "${AGTC_GST_ROOT}/include/gstreamer-1.0" NO_DEFAULT_PATH REQUIRED)
            find_path(AGTC_GLIB_INCLUDE glib.h PATHS "${AGTC_GST_ROOT}/include/glib-2.0" NO_DEFAULT_PATH REQUIRED)
            find_path(AGTC_GLIB_CONFIG glibconfig.h PATHS "${AGTC_GST_ROOT}/lib/glib-2.0/include" NO_DEFAULT_PATH REQUIRED)
            foreach(lib gstreamer-1.0 gstapp-1.0 gstvideo-1.0 gobject-2.0 glib-2.0)
                find_library(AGTC_LIB_${lib} NAMES ${lib} PATHS "${AGTC_GST_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
                target_link_libraries(agtc_gst PUBLIC ${AGTC_LIB_${lib}})
            endforeach()
            target_include_directories(agtc_gst PUBLIC ${AGTC_GST_INCLUDE} ${AGTC_GLIB_INCLUDE} ${AGTC_GLIB_CONFIG})
        else()
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(AGTC_GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0)
            target_link_libraries(agtc_gst PUBLIC PkgConfig::AGTC_GST)
        endif()
        add_executable(test_gst_capture test_gst_capture.cpp)
        target_link_libraries(test_gst_capture PRIVATE agtc_gst)
        add_test(NAME gst_capture COMMAND test_gst_capture)
    endif()
endif()

if(AGTC_WITH_HTTP_SERVER)
    if(NOT EXISTS "${AGTC_HTTPLIB_INCLUDE_DIR}/httplib.h")
        message(FATAL_ERROR "Provide AGTC_HTTPLIB_INCLUDE_DIR with cpp-httplib v0.18.3")
    endif()
    file(SHA256 "${AGTC_HTTPLIB_INCLUDE_DIR}/httplib.h" AGTC_HTTPLIB_SHA)
    if(NOT AGTC_HTTPLIB_SHA STREQUAL "a0a0c13dc086663863dbe6730a19716f7d3744904b6205496e8301750814dbd5")
        message(FATAL_ERROR "cpp-httplib header SHA-256 does not match pinned v0.18.3")
    endif()
    add_library(agtc_http_server STATIC depth_http_server.cpp)
    target_compile_features(agtc_http_server PUBLIC cxx_std_17)
    target_include_directories(agtc_http_server PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${AGTC_HTTPLIB_INCLUDE_DIR})
    target_link_libraries(agtc_http_server PUBLIC agtc_depth)
    if(WIN32)
        target_link_libraries(agtc_http_server PUBLIC ws2_32)
    endif()
    if(AGTC_WITH_PREVIEW)
        find_path(AGTC_TURBOJPEG_INCLUDE turbojpeg.h HINTS "${AGTC_JPEG_ROOT}/include" REQUIRED)
        find_library(AGTC_TURBOJPEG_LIBRARY NAMES turbojpeg HINTS "${AGTC_JPEG_ROOT}/lib" REQUIRED)
        add_library(agtc_preview STATIC depth_preview.cpp stereo_preview.cpp)
        target_compile_features(agtc_preview PUBLIC cxx_std_17)
        target_include_directories(agtc_preview PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${AGTC_TURBOJPEG_INCLUDE} ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(agtc_preview PUBLIC agtc_depth agtc_stereo opencv_core opencv_imgproc ${AGTC_TURBOJPEG_LIBRARY})
        add_executable(test_depth_preview test_depth_preview.cpp)
        target_link_libraries(test_depth_preview PRIVATE agtc_preview)
        add_test(NAME depth_preview COMMAND test_depth_preview)
        add_executable(test_stereo_preview test_stereo_preview.cpp)
        target_link_libraries(test_stereo_preview PRIVATE agtc_preview)
        add_test(NAME stereo_preview COMMAND test_stereo_preview)
        add_executable(benchmark_preview_fixture benchmark_preview_fixture.cpp)
        target_link_libraries(benchmark_preview_fixture PRIVATE agtc_preview)
        add_executable(benchmark_stereo_preview_fixture benchmark_stereo_preview_fixture.cpp)
        target_link_libraries(benchmark_stereo_preview_fixture PRIVATE agtc_preview)
    endif()
    add_executable(test_depth_http_server test_depth_http_server.cpp)
    target_link_libraries(test_depth_http_server PRIVATE agtc_http_server)
    add_test(NAME depth_http_server COMMAND test_depth_http_server)
    if(AGTC_WITH_GSTREAMER)
        add_executable(vision_candidate_main vision_candidate_main.cpp)
        target_link_libraries(vision_candidate_main PRIVATE agtc_gst agtc_http_server)
        add_executable(test_vision_service_fixture test_vision_service_fixture.cpp)
        target_link_libraries(test_vision_service_fixture PRIVATE agtc_gst agtc_http_server)
        if(AGTC_WITH_PREVIEW)
            target_compile_definitions(vision_candidate_main PRIVATE AGTC_WITH_PREVIEW=1)
            target_compile_definitions(test_vision_service_fixture PRIVATE AGTC_WITH_PREVIEW=1)
            target_link_libraries(vision_candidate_main PRIVATE agtc_preview)
            target_link_libraries(test_vision_service_fixture PRIVATE agtc_preview)
        endif()
    endif()
endif()
