cmake_minimum_required(VERSION 3.22)
project(RoboticsLabUWB LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
set(UWB_SOURCE_DIR "" CACHE PATH "Pinned upstream plugin sources (required)")
if(NOT EXISTS "${UWB_SOURCE_DIR}/UWBGazeboSystem.cpp")
  message(FATAL_ERROR "Pass -DUWB_SOURCE_DIR=/path/to/UWBPX4Sim/uwb_gazebo_plugin")
endif()
add_library(UWBGazeboPlugin SHARED ${UWB_SOURCE_DIR}/UWBGazeboSystem.cpp)
target_include_directories(UWBGazeboPlugin PRIVATE ${UWB_SOURCE_DIR})
target_link_libraries(UWBGazeboPlugin PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 gz-transport13::gz-transport13)
